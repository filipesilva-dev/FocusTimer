> Why do I have a folder named ".expo" in my project?

The ".expo" folder is created when an Expo project is started using "expo start" command.

> What do the files contain?

- "devices.json": contains information about devices that have recently opened this project. This is used to populate the "Development sessions" list in your development builds.
- "settings.json": contains the server configuration that is used to serve the application manifest.
- "dev/logs/": contains structured JSONL event logs from CLI commands (e.g. start.log, export.log). These are truncated on each run.

> Should I commit the ".expo" folder?

No, you should not share the ".expo" folder. It does not contain any information that is relevant for other developers working on the project, it is specific to your machine.
Upon project creation, the ".expo" folder is already added to your ".gitignore" file.

## Sobre o projeto

O **FocusTimer** é um aplicativo de organização e produtividade para estudos que aplica a **Técnica Pomodoro** para auxiliar na manutenção do foco em sessões de estudo cronometradas, com intervalos programados.

O projeto está sendo desenvolvido como trabalho final da disciplina de **Desenvolvimento Mobile com React Native**, com o objetivo de aplicar na prática os conceitos vistos em sala: navegação, hooks, componentização, ciclo de vida, notificações, requisições a Web Services e persistência de dados.

A proposta vai além de um cronômetro simples: o app permitirá que o usuário organize tarefas, crie coleções de flashcards, acompanhe o progresso semanal e receba notificações para manter a constância nos estudos.

file_ext() {
  local name; name="$(basename "$1")"
  if [[ "$name" == *.* ]]; then
    printf '%s' "${name##*.}" | tr '[:upper:]' '[:lower:]'
  fi
}

should_skip() {
  local path="$1" base ext
  base="$(basename "$path")"
  ext="$(file_ext "$path")"

  [[ "$base" == "$OUTPUT_BASE" ]] && return 0

  for f in "${SKIP_FILES[@]}"; do
    [[ "$base" == "$f" ]] && return 0
  done
  for e in "${SKIP_EXT[@]}"; do
    [[ "$ext" == "$e" ]] && return 0
  done
  return 1

info "Varrendo projeto em $ROOT"

prune=()
for d in "${SKIP_DIRS[@]}"; do
  [[ ${#prune[@]} -gt 0 ]] && prune+=( -o )

FILES_OK="$(mktemp)"
trap 'rm -f "$FILES_RAW" "$FILES_OK"' EXIT

while IFS= read -r -d '' raw; do
  rel="${raw#./}"

done < "$FILES_RAW"

    n=$((n + 1))
    printf '%4d. %s\n' "$n" "$f"
  done < "$FILES_OK"
  echo ""

  echo "───────────────────────────────────────────────────────────────────"
  echo "ESTRUTURA DE PASTAS"
  echo "───────────────────────────────────────────────────────────────────"

      indent=""

      printf '%s%s/\n' "$indent" "$(basename "$p")"
    done < <(find . -type d \( "${prune[@]}" \) -prune -o -type d -print | sort)

  echo ""

  while IFS= read -r file; do
    [[ -z "$file" ]] && continue
    [[ ! -f "$file" ]] && continue

    size=$(wc -c < "$file" | tr -d ' ')
    lines=$(wc -l < "$file" | tr -d ' ')
    ext="$(file_ext "$file")"

    echo "───────────────────────────────────────────────────────────────────"
    echo "FILE: $file"
    echo "LINHAS: $lines · BYTES: $size"

    if (( size > MAX_BYTES )); then

      continue
    fi

    if [[ -n "$ext" ]]; then
      printf '```%s\n' "$ext"
    else
      printf '```\n'
    fi

    cat "$file"
    last="$(tail -c1 "$file" 2>/dev/null || true)"
    [[ "$last" != $'\n' ]] && echo ""
    printf '```\n\n'
  done < "$FILES_OK"

  echo "═══════════════════════════════════════════════════════════════════"
  echo "FIM"
  echo "═══════════════════════════════════════════════════════════════════"

  }, [colecaoId, atualizacao]);

  function fechar() {
    if (salvando.current || exclusaoEmAndamento.current) return;
    if (revisaoAberta) {
      voltarRevisao.current?.();
      return;
    }
    if (formularioAberto) setFormularioAberto(false);
    else onFechar();
  }

  function abrirFormulario(card: Card | null) {
    setCardEditando(card);
    setFormularioAberto(true);
  }

  async function salvar(dados: DadosCard) {
    let atualizado = true;
    try {
      if (cardEditando) atualizado = await atualizarCard(colecaoId, cardEditando.id, dados);
      else await criarCard(colecaoId, dados);
    } catch {
      throw new Error('Não foi possível salvar o flashcard. Confira a coleção e tente novamente.');
    }
    if (!atualizado) throw new Error('Este flashcard não existe mais. Cancele a edição e atualize a lista.');
    if (montada.current) setAtualizacao((valor) => valor + 1);
  }

  async function remover(card: Card) {
    if (exclusaoEmAndamento.current || !montada.current) return;
    exclusaoEmAndamento.current = true;
    setExcluindo(true);
    try {
      const removido = await excluirCard(colecaoId, card.id);
      if (montada.current) {
        setAtualizacao((valor) => valor + 1);
        if (!removido) mostrarAviso('Flashcard não encontrado', 'A lista será atualizada.');
      }
    } catch {
      if (montada.current) {
        setAtualizacao((valor) => valor + 1);
        mostrarAviso('Não foi possível confirmar a exclusão', 'A lista será atualizada. Confira e tente novamente.');
      }
    } finally {
      exclusaoEmAndamento.current = false;
      if (montada.current) setExcluindo(false);
    }
  }

  function confirmarExclusao(card: Card) {
    confirmarAcao({
      titulo: 'Excluir flashcard?',
      mensagem: 'Deseja excluir este flashcard da coleção?',
      onConfirmar: () => { void remover(card); },
    });
  }

  return (
    <Modal visible animationType="slide" onRequestClose={fechar}>
      <SafeAreaView style={styles.safeArea}>
        {revisaoAberta ? (
          <Revisao cards={cards} controleVoltar={voltarRevisao} onVoltar={() => setRevisaoAberta(false)} />
        ) : formularioAberto ? (
          <CardForm card={cardEditando} onSalvar={salvar} onCancelar={fechar}
            onOcupado={(valor) => { salvando.current = valor; }} />
        ) : (
          <View style={styles.container}>
            <Pressable onPress={fechar} disabled={excluindo} accessibilityRole="button"
              accessibilityLabel="Voltar às coleções" accessibilityState={{ disabled: excluindo }}
              style={[styles.voltar, excluindo && styles.desabilitado]}>
              <Text style={styles.textoVoltar}>‹ Voltar às coleções</Text>
            </Pressable>
            <Text style={styles.titulo} accessibilityRole="header">{colecao?.nome ?? 'Flashcards'}</Text>
            <Text style={styles.contagem}>{cards.length} {cards.length === 1 ? 'flashcard' : 'flashcards'}</Text>
            <Pressable onPress={() => abrirFormulario(null)} disabled={carregando || excluindo || !colecao}
              accessibilityRole="button" accessibilityLabel="Adicionar flashcard"
              accessibilityState={{ disabled: carregando || excluindo || !colecao }}
              style={[styles.botao, (carregando || excluindo || !colecao) && styles.desabilitado]}>
              <Text style={styles.textoBotao}>Adicionar flashcard</Text>
            </Pressable>
            <Pressable onPress={() => setRevisaoAberta(true)}
              disabled={carregando || excluindo || !!erro || cards.length === 0 || !colecao}
              accessibilityRole="button" accessibilityLabel="Revisar flashcards desta coleção"
              accessibilityState={{ disabled: carregando || excluindo || !!erro || cards.length === 0 || !colecao }}
              style={[styles.botao, styles.revisar,
                (carregando || excluindo || !!erro || cards.length === 0 || !colecao) && styles.desabilitado]}>
              <Text style={styles.textoVoltar}>Revisar flashcards</Text>
            </Pressable>
            {carregando ? (
              <View style={styles.estado}>
                <ActivityIndicator size="large" color={colors.primaryDark} />

                </Pressable>
              </View>
            ) : (

import type { Card, Colecao, DadosCard, DadosColecao } from '../types/cards';
import { validarCard, validarColecao, validarId } from '../utils/validacao';
import { obterBanco } from './connection';

const CONSULTA_COLECOES = `
  SELECT c.id, c.nome, c.descricao, c.criada_em AS criadaEm,
         COUNT(card.id) AS totalCards
  FROM colecoes c
  LEFT JOIN cards card ON card.colecao_id = c.id
`;

export async function listarColecoes(): Promise<Colecao[]> {
  const banco = await obterBanco();
  return banco.getAllAsync<Colecao>(`
    ${CONSULTA_COLECOES}
    GROUP BY c.id
    ORDER BY c.criada_em ASC, c.id ASC
  `);
}

export async function obterColecao(id: number): Promise<Colecao | null> {
  validarId(id);
  const banco = await obterBanco();
  return banco.getFirstAsync<Colecao>(
    `${CONSULTA_COLECOES} WHERE c.id = ? GROUP BY c.id`,
    [id],
  );
}

export async function criarColecao(dados: DadosColecao): Promise<number> {
  const { nome, descricao } = validarColecao(dados);
  const banco = await obterBanco();
  const resultado = await banco.runAsync(
    'INSERT INTO colecoes (nome, descricao, criada_em) VALUES (?, ?, ?)',
    [nome, descricao, new Date().toISOString()],
  );
  return resultado.lastInsertRowId;
}

export async function atualizarColecao(id: number, dados: DadosColecao): Promise<boolean> {
  validarId(id);
  const { nome, descricao } = validarColecao(dados);
  const banco = await obterBanco();
  const resultado = await banco.runAsync(
    'UPDATE colecoes SET nome = ?, descricao = ? WHERE id = ?',
    [nome, descricao, id],
  );
  return resultado.changes > 0;
}

export async function excluirColecao(id: number): Promise<boolean> {
  validarId(id);
  const banco = await obterBanco();
  const resultado = await banco.runAsync('DELETE FROM colecoes WHERE id = ?', [id]);
  return resultado.changes > 0;
}

  const resultado = await banco.runAsync(

    [colecaoId, pergunta, resposta, dificuldade, agora, agora],

import { openDatabaseAsync, type SQLiteDatabase } from 'expo-sqlite';
import { Platform } from 'react-native';

const NOME_BANCO = 'focustimer.db';
let conexao: Promise<SQLiteDatabase> | null = null;

async function abrirBanco(): Promise<SQLiteDatabase> {
  if (Platform.OS === 'web' && (!globalThis.isSecureContext || !navigator.storage?.getDirectory)) {
    throw new Error('Use localhost ou HTTPS em um navegador com suporte a armazenamento local.');
  }
  const banco = await openDatabaseAsync(NOME_BANCO);
  try {
    await banco.execAsync(`
      PRAGMA foreign_keys = ON;

      CREATE TABLE IF NOT EXISTS colecoes (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        nome TEXT NOT NULL,
        descricao TEXT,
        criada_em TEXT NOT NULL
      );

      CREATE TABLE IF NOT EXISTS cards (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        colecao_id INTEGER NOT NULL REFERENCES colecoes(id) ON DELETE CASCADE,
        pergunta TEXT NOT NULL,
        resposta TEXT NOT NULL,
        dificuldade TEXT NOT NULL DEFAULT 'medio',
        criado_em TEXT NOT NULL,
        atualizado_em TEXT NOT NULL
      );
    `);
    return banco;
  } catch (erro) {
    await banco.closeAsync().catch(() => undefined);
    throw erro;
  }
}

export function obterBanco(): Promise<SQLiteDatabase> {
  if (!conexao) {
    conexao = abrirBanco().catch((erro: unknown) => {
      // Uma falha temporária não deve impedir todas as próximas tentativas.
      conexao = null;
      throw erro;
    });
  }
  return conexao;
}

import React, { useEffect, useState } from 'react';

  View, Text, StyleSheet, Pressable, Modal, TextInput, ScrollView,
  KeyboardAvoidingView, Platform,
} from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { colors } from '../../theme/colors';

type Props = {
  visible: boolean;
  tarefa: Task | null; // null = tarefa nova
  data: string; // dia selecionado na agenda (AAAA-MM-DD)
  onFechar: () => void;
  onSalvar: (tarefa: Task) => void;
  onExcluir: (id: string) => void;
};

const OPCOES_PRIORIDADE = [
  { valor: 1, label: 'Baixa' },
  { valor: 2, label: 'Média' },
  { valor: 3, label: 'Alta' },
];

export default function TarefaModal({ visible, tarefa, data, onFechar, onSalvar, onExcluir }: Props) {
  const [titulo, setTitulo] = useState('');
  const [materia, setMateria] = useState('');
  const [prioridade, setPrioridade] = useState(2);
  const [pomodoros, setPomodoros] = useState(1);
  const [feitos, setFeitos] = useState(0);
  const [notas, setNotas] = useState('');
  const [erro, setErro] = useState('');
  const [confirmarExclusao, setConfirmarExclusao] = useState(false);

  // toda vez que abre, preenche com a tarefa (editar) ou limpa (nova)
  useEffect(() => {
    if (!visible) return;
    setTitulo(tarefa ? tarefa.title : '');
    setMateria(tarefa ? tarefa.className : '');
    setPrioridade(tarefa ? tarefa.priority : 2);
    setPomodoros(tarefa ? tarefa.pomodoroAmount : 1);
    setFeitos(tarefa ? tarefa.completedPomodoros : 0);
    setNotas(tarefa ? tarefa.notes : '');
    setErro('');
    setConfirmarExclusao(false);
  }, [visible, tarefa]);

  function mudarPomodoros(delta: number) {
    const novo = Math.max(1, pomodoros + delta);
    setPomodoros(novo);
    if (feitos > novo) setFeitos(novo);
  }

  function mudarFeitos(delta: number) {
    setFeitos(Math.min(pomodoros, Math.max(0, feitos + delta)));
  }

  function salvar() {
    if (titulo.trim() === '') {
      setErro('Digite um título para a tarefa.');
      return;
    }

    const dados = {
      title: titulo.trim(),
      className: materia.trim() || 'Geral',
      priority: prioridade,
      pomodoroAmount: pomodoros,
      completedPomodoros: feitos,
      notes: notas.trim(),
    };

    if (tarefa) {
      onSalvar({ ...tarefa, ...dados });
    } else {
      onSalvar(createTask({ ...dados, date: data, status: false }));
    }
  }

  function excluir() {
    if (!tarefa) return;
    // primeiro clique pede confirmação, o segundo exclui
    if (!confirmarExclusao) {
      setConfirmarExclusao(true);
      return;
    }
    onExcluir(tarefa.id);
  }

  return (
    <Modal visible={visible} transparent animationType="fade" onRequestClose={onFechar}>
      <KeyboardAvoidingView
        style={styles.fundo}
        behavior={Platform.OS === 'ios' ? 'padding' : undefined}
      >
        <View style={styles.caixa}>
          <View style={styles.header}>
            <Text style={styles.titulo}>{tarefa ? 'Editar tarefa' : 'Nova tarefa'}</Text>

            </Pressable>
          </View>

            <Text style={styles.label}>Título</Text>
            <TextInput
              style={styles.input}
              value={titulo}
              onChangeText={(texto) => {
                setTitulo(texto);
                setErro('');
              }}
              placeholder="Ex: Revisar capítulo 3"
              placeholderTextColor={colors.textLight}
            />
            {erro !== '' && <Text style={styles.erro}>{erro}</Text>}

            <Text style={styles.label}>Matéria</Text>
            <TextInput
              style={styles.input}
              value={materia}
              onChangeText={setMateria}
              placeholder="Ex: Matemática"
              placeholderTextColor={colors.textLight}
            />

            <Text style={styles.label}>Prioridade</Text>
            <View style={styles.linhaBotoes}>
              {OPCOES_PRIORIDADE.map((opcao) => (
                <Pressable
                  key={opcao.valor}
                  onPress={() => setPrioridade(opcao.valor)}
                  style={[styles.opcao, prioridade === opcao.valor && styles.opcaoAtiva]}
                >
                  <Text style={[styles.opcaoTexto, prioridade === opcao.valor && styles.opcaoTextoAtivo]}>
                    {opcao.label}
                  </Text>
                </Pressable>
              ))}
            </View>

            <View style={styles.linhaBotoes}>
              <View style={{ flex: 1 }}>
                <Text style={styles.label}>Pomodoros planejados</Text>
                <View style={styles.contador}>
                  <Pressable onPress={() => mudarPomodoros(-1)} style={styles.botaoContador} accessibilityLabel="Diminuir pomodoros planejados">
                    <Ionicons name="remove" size={18} color={colors.primaryDark} />
                  </Pressable>
                  <Text style={styles.valorContador}>{pomodoros}</Text>
                  <Pressable onPress={() => mudarPomodoros(1)} style={styles.botaoContador} accessibilityLabel="Aumentar pomodoros planejados">
                    <Ionicons name="add" size={18} color={colors.primaryDark} />
                  </Pressable>
                </View>
              </View>

              <View style={{ flex: 1 }}>
                <Text style={styles.label}>Feitos</Text>
                <View style={styles.contador}>
                  <Pressable onPress={() => mudarFeitos(-1)} style={styles.botaoContador} accessibilityLabel="Diminuir pomodoros feitos">
                    <Ionicons name="remove" size={18} color={colors.primaryDark} />
                  </Pressable>
                  <Text style={styles.valorContador}>{feitos}</Text>
                  <Pressable onPress={() => mudarFeitos(1)} style={styles.botaoContador} accessibilityLabel="Aumentar pomodoros feitos">
                    <Ionicons name="add" size={18} color={colors.primaryDark} />
                  </Pressable>
                </View>
              </View>
            </View>

            <Text style={styles.label}>Anotações</Text>

  const [erro, setErro] = useState('');
  const [atualizacao, setAtualizacao] = useState(0);
  const [formularioAberto, setFormularioAberto] = useState(false);
  const [colecaoEditando, setColecaoEditando] = useState<Colecao | null>(null);

    return () => { montada.current = false; };

    let ativa = true;
    async function carregar() {
      setCarregando(true);
      setErro('');
      try {
        const dados = await listarColecoes();
        if (ativa) setColecoes(dados);
      } catch {
        if (ativa) setErro('Não foi possível carregar suas coleções.');
      } finally {
        if (ativa) setCarregando(false);
      }
    }
    void carregar();
    return () => { ativa = false; };
  }, [atualizacao]));

  function abrirFormulario(colecao: Colecao | null) {
    setColecaoEditando(colecao);
    setFormularioAberto(true);

    let atualizada = true;
    try {
      if (colecaoEditando) {
        atualizada = await atualizarColecao(colecaoEditando.id, dados);
      } else {
        await criarColecao(dados);
      }
    } catch {
      throw new Error('Não foi possível salvar a coleção. Seus campos foram mantidos para tentar novamente.');
    }
    if (!atualizada) throw new Error('Esta coleção não existe mais. Feche o formulário e atualize a lista.');
    if (montada.current) {
      setAtualizacao((valor) => valor + 1);
    }
  }

    } finally {

    }
  }

    });
  }

      <View style={styles.conteudo}>
        <Text style={styles.titulo} accessibilityRole="header">Minhas coleções</Text>

        </Pressable>
        {carregando ? (

            <ActivityIndicator size="large" color={colors.primaryDark} />

          </View>
        ) : erro ? (
          <View style={styles.estado}>

            <Pressable onPress={() => setAtualizacao((valor) => valor + 1)}
              accessibilityRole="button" style={styles.botao}>
              <Text style={styles.textoBotao}>Tentar novamente</Text>
            </Pressable>
          </View>
        ) : (
          <FlatList data={colecoes} keyExtractor={(item) => String(item.id)}
            contentContainerStyle={styles.lista}
            renderItem={({ item }) => (
              <ColecaoItem colecao={item} desabilitado={excluindo}
                onAbrir={() => setColecaoAberta(item)}
                onEditar={() => abrirFormulario(item)} onExcluir={() => confirmarExclusao(item)} />
            )}
            ListEmptyComponent={<Text style={styles.mensagem}>Nenhuma coleção ainda. Toque em “Nova coleção” para começar.</Text>} />
        )}
      </View>
      {formularioAberto && <ColecaoForm colecao={colecaoEditando}
        onCancelar={() => setFormularioAberto(false)} onSalvar={salvar} />}
      {colecaoAberta && <ColecaoCards key={colecaoAberta.id} colecaoId={colecaoAberta.id}
        onFechar={() => {
          setColecaoAberta(null);
          setAtualizacao((valor) => valor + 1);
        }} />}
    </View>
  );
}

function ConfigTempo({ label, segundos, onChange }: { label: string; segundos: number; onChange: (seg: number) => void }) {
  const hms = segundosParaHMS(segundos);
  const [campos, setCampos] = useState({ h: String(hms.h), m: String(hms.m), s: String(hms.s) });
  const [atualizado, setAtualizado] = useState(false);

  useEffect(() => {
    if (!atualizado) return;
    const novo = segundosParaHMS(segundos);
    setCampos({ h: String(novo.h), m: String(novo.m), s: String(novo.s) });
    setAtualizado(false);
  }, [segundos, atualizado]);

  function alterar(campo: 'h' | 'm' | 's', texto: string) {
    const clean = texto.replace(/[^0-9]/g, '');
    const novos = { ...campos, [campo]: clean };
    setCampos(novos);
    const hN = parseInt(novos.h, 10) || 0;
    const mN = parseInt(novos.m, 10) || 0;
    const sN = parseInt(novos.s, 10) || 0;
    setAtualizado(true);
    onChange(Math.max(1, hmsParaSegundos(hN, mN, sN)));
  }

  return (
    <View style={styles.configTempoBloco}>
      <View style={styles.configHeader}>
        <Text style={styles.configTempoLabel}>{label}</Text>
        <Text style={styles.configTempoPreview}>Total: {formatarTempoCurto(segundos)}</Text>
      </View>
      <View style={styles.unitsRow}>
        <View style={styles.unitCol}>
          <TextInput style={styles.unitInput} keyboardType="number-pad" value={campos.h} onChangeText={(t) => alterar('h', t)} selectTextOnFocus maxLength={3} />
          <Text style={styles.unitLabel}>h</Text>
        </View>

import AsyncStorage from '@react-native-async-storage/async-storage';
import { Task } from '../types';

const TASKS_KEY = '@FocusTimer/tasks';

export function isTask(value: unknown): value is Task {
  if (typeof value !== 'object' || value === null) return false;
  const task = value as Record<string, unknown>;
  return (
    typeof task.id === 'string' &&
    typeof task.title === 'string' &&
    typeof task.className === 'string' &&
    typeof task.priority === 'number' &&
    typeof task.pomodoroAmount === 'number' &&
    typeof task.completedPomodoros === 'number' &&
    typeof task.date === 'string' &&
    typeof task.status === 'boolean' &&
    typeof task.notes === 'string'
  );
}

export async function loadTasks(): Promise<Task[]> {
  try {
    const saved = await AsyncStorage.getItem(TASKS_KEY);
    if (saved === null) return [];
    const parsed: unknown = JSON.parse(saved);
    if (!Array.isArray(parsed) || !parsed.every(isTask)) {
      console.warn('Saved tasks have an invalid format');
      return [];
    }
    return parsed;
  } catch (e) {
    console.error('Failed to load tasks:', e);
    return [];
  }
}

export async function saveTasks(tasks: Task[]): Promise<void> {
  try {
    await AsyncStorage.setItem(TASKS_KEY, JSON.stringify(tasks));
  } catch (e) {
    console.error('Failed to save tasks:', e);
  }
}

export function createTask(task: Omit<Task, "id">): Task {
  return {
    ...task,
    id: `${Date.now()}-${Math.random().toString(36).slice(2, 10)}`,
  };
}

import type { Card, Dificuldade } from '../types/cards';

export type NivelRevisao = 'todos' | Dificuldade;
export type OrdemRevisao = 'sequencial' | 'aleatoria';

export interface EstadoRevisao {
  cards: Card[];
  fila: Card[];
  respostaVisivel: boolean;
  tentativas: number;
  acertosPrimeira: number;
  errosIds: number[];
  encerrada: boolean;
}

export function filtrarCards(cards: Card[], nivel: NivelRevisao): Card[] {
  return cards.filter((card) => nivel === 'todos' || card.dificuldade === nivel);
}

export function embaralharCards(cards: Card[], aleatorio: () => number = Math.random): Card[] {
  const resultado = [...cards];
  for (let i = resultado.length - 1; i > 0; i--) {
    const j = Math.floor(aleatorio() * (i + 1));
    [resultado[i], resultado[j]] = [resultado[j], resultado[i]];
  }
  return resultado;
}

export function iniciarRevisao(
  cards: Card[], nivel: NivelRevisao, ordem: OrdemRevisao, aleatorio: () => number = Math.random,
): EstadoRevisao {
  const selecionados = filtrarCards(cards, nivel)
    .map((card) => ({ ...card }))
    .sort((a, b) => a.criadoEm.localeCompare(b.criadoEm) || a.id - b.id);
  if (selecionados.length === 0) {
    throw new Error('Nenhum flashcard corresponde à dificuldade escolhida.');
  }
  return {
    cards: selecionados,
    fila: ordem === 'aleatoria' ? embaralharCards(selecionados, aleatorio) : [...selecionados],
    respostaVisivel: false,
    tentativas: 0,
    acertosPrimeira: 0,
    errosIds: [],
    encerrada: false,
  };
}

export function revelarResposta(estado: EstadoRevisao, cardId: number): EstadoRevisao {
  if (estado.encerrada || estado.fila[0]?.id !== cardId) return estado;
  return { ...estado, respostaVisivel: true };
}

export function responderCard(estado: EstadoRevisao, cardId: number, acertou: boolean): EstadoRevisao {
  const card = estado.fila[0];
  if (estado.encerrada || !estado.respostaVisivel || !card || card.id !== cardId) return estado;
  const jaErrou = estado.errosIds.includes(card.id);
  const restante = estado.fila.slice(1);
  const fila = acertou ? restante : [...restante, card];
  return {
    ...estado,
    fila,
    respostaVisivel: false,
    tentativas: estado.tentativas + 1,
    acertosPrimeira: estado.acertosPrimeira + (acertou && !jaErrou ? 1 : 0),
    errosIds: !acertou && !jaErrou ? [...estado.errosIds, card.id] : estado.errosIds,
    encerrada: fila.length === 0,
  };
}

export function encerrarRevisao(estado: EstadoRevisao): EstadoRevisao {
  return { ...estado, encerrada: true, respostaVisivel: false };
}

export function resumirRevisao(estado: EstadoRevisao) {
  return {
    total: estado.cards.length,
    concluidos: estado.cards.length - estado.fila.length,
    pendentes: estado.fila.length,
    tentativas: estado.tentativas,
    acertosPrimeira: estado.acertosPrimeira,
    cardsComErro: estado.cards.filter((card) => estado.errosIds.includes(card.id)),
  };
}

export type HMS = { h: number; m: number; s: number };

export function segundosParaHMS(total: number): HMS {
  const safe = Math.max(0, Math.floor(total));
  return {
    h: Math.floor(safe / 3600),
    m: Math.floor((safe % 3600) / 60),
    s: safe % 60,
  };
}

export function hmsParaSegundos(h: number, m: number, s: number): number {
  const hh = Math.max(0, Math.floor(h || 0));
  const mm = Math.max(0, Math.floor(m || 0));
  const ss = Math.max(0, Math.floor(s || 0));
  return hh * 3600 + mm * 60 + ss;
}

export function formatarTempoCurto(total: number): string {
  if (!total || total <= 0) return '0s';
  const { h, m, s } = segundosParaHMS(total);
  const partes: string[] = [];
  if (h > 0) partes.push(`${h}h`);
  if (m > 0) partes.push(`${m}min`);
  if (s > 0) partes.push(`${s}s`);
  return partes.join(' ');
}

export function formatarTimer(total: number): string {
  const safe = Math.max(0, Math.floor(total));
  const { h, m, s } = segundosParaHMS(safe);
  const mm = String(m).padStart(2, '0');
  const ss = String(s).padStart(2, '0');
  if (h > 0) return `${String(h).padStart(2, '0')}:${mm}:${ss}`;
  return `${mm}:${ss}`;
}

import type { DadosCard, DadosColecao, Dificuldade } from '../types/cards';

export function validarId(id: number): void {
  if (!Number.isSafeInteger(id) || id <= 0) {
    throw new Error('O identificador deve ser um número inteiro positivo.');
  }
}

export function validarColecao(dados: DadosColecao): {
  nome: string;
  descricao: string | null;
} {
  if (!dados || typeof dados.nome !== 'string') {
    throw new Error('Informe o nome da coleção.');
  }

  const nome = dados.nome.trim();
  if (Array.from(nome).length < 1 || Array.from(nome).length > 60) {
    throw new Error('O nome deve ter entre 1 e 60 caracteres.');
  }

  if (dados.descricao != null && typeof dados.descricao !== 'string') {
    throw new Error('A descrição deve ser um texto.');
  }

  const descricao = dados.descricao?.trim() || null;
  if (descricao !== null && Array.from(descricao).length > 200) {
    throw new Error('A descrição deve ter até 200 caracteres.');
  }

  return { nome, descricao };
}

export function validarCard(dados: DadosCard): {
  pergunta: string;
  resposta: string;
  dificuldade: Dificuldade;
} {
  if (!dados || typeof dados.pergunta !== 'string' || typeof dados.resposta !== 'string') {
    throw new Error('Informe a pergunta e a resposta do flashcard.');
  }
  const pergunta = dados.pergunta.trim();
  const resposta = dados.resposta.trim();
  if (Array.from(pergunta).length < 1 || Array.from(pergunta).length > 500) {
    throw new Error('A pergunta deve ter entre 1 e 500 caracteres.');
  }
  if (Array.from(resposta).length < 1 || Array.from(resposta).length > 500) {
    throw new Error('A resposta deve ter entre 1 e 500 caracteres.');
  }
  const dificuldade = dados.dificuldade === undefined ? 'medio' : dados.dificuldade;
  if (dificuldade !== 'facil' && dificuldade !== 'medio' && dificuldade !== 'dificil') {
    throw new Error('Escolha uma dificuldade válida.');
  }
  return { pergunta, resposta, dificuldade };
}
